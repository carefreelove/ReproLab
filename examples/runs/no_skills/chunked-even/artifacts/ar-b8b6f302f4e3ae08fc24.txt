4637:     [[1, 2, 3], [4, 5, 6], [7]]

4640:     iterator = iter(iterable)
4641: 
4642:     # Initialize a buffer to process the chunks while keeping
4643:     # some back to fill any underfilled chunks
4644:     min_buffer = (n - 1) * (n - 2)
4645:     buffer = list(islice(iterator, min_buffer))
4646:

4648:     for _ in islice(map(buffer.append, iterator), n, None, n):
4649:         yield buffer[:n]

4662:     partial_size = full_size - 1
4663:     num_full = length - partial_size * num_lists

4666:     partial_start_idx = num_full * full_size
4667:     if full_size > 0:

4669:             yield buffer[i : i + full_size]

4673:         for i in range(partial_start_idx, length, partial_size):
